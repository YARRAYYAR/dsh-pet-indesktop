import hashlib,json,plistlib,subprocess,time,zipfile
from pathlib import Path

archive=Path('/Users/ray/Downloads/seeky-pet-4.2.1.7-macOS-arm64.zip')
assert source.exists() and not archive.exists()
cmd=['ditto','-c','-k','--sequesterRsrc','--keepParent',str(source),str(archive)]
